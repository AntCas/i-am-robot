Listen for the robot-verification-passed event and send the returned resultToken to your backend.

Frontend

document.addEventListener("robot-verification-passed", async (event) => {
  const { resultToken } = event.detail;

  const response = await fetch("/your-backend/verify-robot", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ resultToken }),
  });

  const verification = await response.json();
  if (verification.success) {
    // Continue with the protected action.
  }
});
import { useEffect } from "react";

export function RobotVerificationListener({ onVerified }) {
  useEffect(() => {
    async function handleVerificationPassed(event) {
      const { resultToken } = event.detail;

      const response = await fetch("/your-backend/verify-robot", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ resultToken }),
      });

      const verification = await response.json();
      if (verification.success) {
        onVerified?.(verification);
      }
    }

    document.addEventListener("robot-verification-passed", handleVerificationPassed);
    return () => {
      document.removeEventListener("robot-verification-passed", handleVerificationPassed);
    };
  }, [onVerified]);

  return null;
}

Backend

import express from "express";

const app = express();
app.use(express.json());

app.post("/your-backend/verify-robot", async (request, response) => {
  const { resultToken } = request.body;

  const robotResponse = await fetch("https://castrio.me/im-a-robot/api/verify", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      secret: process.env.I_AM_ROBOT_SITE_SECRET,
      resultToken,
    }),
  });

  response.status(robotResponse.status).json(await robotResponse.json());
});
from flask import Flask, jsonify, request
import os
import requests

app = Flask(__name__)

@app.post("/your-backend/verify-robot")
def verify_robot():
    result_token = request.json["resultToken"]

    response = requests.post(
        "https://castrio.me/im-a-robot/api/verify",
        json={
            "secret": os.environ["I_AM_ROBOT_SITE_SECRET"],
            "resultToken": result_token,
        },
        timeout=10,
    )

    return jsonify(response.json()), response.status_code
